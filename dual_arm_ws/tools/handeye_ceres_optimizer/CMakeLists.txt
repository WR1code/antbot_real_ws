cmake_minimum_required(VERSION 3.16)
project(handeye_ceres_optimizer LANGUAGES CXX)
find_package(Ceres 2.1 REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 CONFIG REQUIRED)
pybind11_add_module(_handeye_ceres src/optimizer.cpp)
target_compile_features(_handeye_ceres PRIVATE cxx_std_17)
target_link_libraries(_handeye_ceres PRIVATE Ceres::ceres)
target_compile_options(_handeye_ceres PRIVATE -Wall -Wextra -Wpedantic)
